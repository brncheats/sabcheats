local function mk(className, props, parent)
    local inst = Instance.new(className)
    for k, v in pairs(props) do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end
local function corner(obj, r) return mk("UICorner", { CornerRadius = UDim.new(0, r) }, obj) end
local function stroke(obj, color, thickness, transparency, mode)
    return mk("UIStroke", {
        Color = color, Thickness = thickness,
        Transparency = transparency or 0,
        ApplyStrokeMode = mode or Enum.ApplyStrokeMode.Border,
    }, obj)
end
local function textOutline(label, transparency)
    return stroke(label, Color3.fromRGB(0, 0, 0), 1, transparency or 0.45, Enum.ApplyStrokeMode.Contextual)
end
local function nseq(a, b)
    return NumberSequence.new({
        NumberSequenceKeypoint.new(0, a),
        NumberSequenceKeypoint.new(1, b),
    })
end
local function gradient(obj, transparency, rotation)
    local g = mk("UIGradient", {}, obj)
    if transparency then g.Transparency = transparency end
    if rotation then g.Rotation = rotation end
    return g
end
local function tween(obj, props, time, style, dir)
    TweenService:Create(
        obj,
        TweenInfo.new(time or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    ):Play()
end
local function clamp(v, lo, hi)
    if v < lo then return lo elseif v > hi then return hi end
    return v
end
local function styleField(obj, radius)
    obj.BorderSizePixel = 0
    obj.BackgroundColor3 = C.field
    corner(obj, radius)
    return stroke(obj, C.line, 1, 0.80)
end
local function addPress(btn, scale)

end

--==============================================================
-- POWER FORMAT
--==============================================================
local SUFFIX = { { "t", 1e12 }, { "b", 1e9 }, { "m", 1e6 }, { "k", 1e3 } }
local MULT   = { k = 1e3, m = 1e6, b = 1e9, t = 1e12 }

local function formatPower(value)
    local n = math.floor(tonumber(value) or 0)
    if n < 1000 then return tostring(n) end
    for _, pair in ipairs(SUFFIX) do
        local tag, size = pair[1], pair[2]
        if n >= size then
            local q = n / size
            local text
            if q >= 100 then text = string.format("%d", math.floor(q))
            else
                text = string.format("%.1f", math.floor(q * 10) / 10)
                if string.find(text, "%.") then text = string.gsub(text, "%.?0+$", "") end
            end
            return text .. tag
        end
    end
end

local function parsePower(value)
    if value == nil then return nil end
    local s = string.lower(tostring(value))
    s = string.gsub(s, "[%s_,]", "")
    local number, tag = string.match(s, "^([%d%.]+)([kmbt]?)$")
    if not number then return nil end
    local n = tonumber(number)
    if not n then return nil end
    if tag ~= "" then n = n * MULT[tag] end
    return math.floor(clamp(n, MIN_POWER, MAX_POWER))
end

local function emptyOnFocus(box, getPlaceholder)
    box.Focused:Connect(function()
        box.PlaceholderText = getPlaceholder()
        box.Text = ""
    end)
end

local function friendlyKey(name)
    local text = string.gsub(name, "^Button", "")
    text = string.gsub(text, "^DPad", "D-")
    text = string.gsub(text, "^Keypad", "NUM")
    if text == "L1" then text = "LB" end
    if text == "L2" then text = "LT" end
    if text == "R1" then text = "RB" end
    if text == "R2" then text = "RT" end
    if text == "Select" then text = "VIEW" end
    if text == "Start" then text = "MENU" end
    if text == "" then return "?" end
    if string.len(text) == 1 then return string.upper(text) end
    return text
end

local function refreshKeyBox()
    if capturing then
        keyBox.Text = "..."
        keyBox.TextColor3 = C.white
        keyBox.BackgroundColor3 = C.fieldRun
        return
    end
    keyBox.BackgroundColor3 = C.field
    if bind.kind == "none" then
        keyBox.Text = "--"
        keyBox.TextColor3 = C.dim
    else
        keyBox.Text = bind.label
        keyBox.TextColor3 = C.white
    end
end

local function setBind(kind, code, label, save)
    bind.kind = kind
    bind.code = code
    bind.label = label
    refreshKeyBox()
    if save ~= false and queueSave then queueSave() end
end

local function inputMatchesBind(input)
    if bind.kind == "none" or capturing then return false end
    if bind.kind == "mouse" then
        local wanted = Enum.UserInputType[bind.code]
        return wanted ~= nil and input.UserInputType == wanted
    end
    if input.KeyCode ~= Enum.KeyCode.Unknown and input.KeyCode.Name == bind.code then
        return true
    end
    return false
end

--==============================================================
-- REMOTE RESOLUTION
--==============================================================
local function tryFindNet()
    local ok, pkgs = pcall(function() return ReplicatedStorage:FindFirstChild("Packages") end)
    if not ok or not pkgs then return nil end
    local netObj = pkgs:FindFirstChild("Net")

    if netObj then
        local okA, candidate = pcall(function() return netObj:GetChildren()[230] end)
        if okA and candidate and candidate:IsA("RemoteEvent") then return candidate end
    end
    if netObj then
        for _, child in ipairs(netObj:GetChildren()) do
            if child:IsA("RemoteEvent") then
                local n = string.lower(child.Name)
                if string.find(n, "net") or string.find(n, "packet") or string.find(n, "remote") then
                    return child
                end
            end
        end
    end
    for _, child in ipairs(pkgs:GetChildren()) do
        if child:IsA("RemoteEvent") then return child end
    end
    for _, child in ipairs(ReplicatedStorage:GetChildren()) do
        if child:IsA("RemoteEvent") then
            local n = string.lower(child.Name)
            if string.find(n, "net") or string.find(n, "packet") then return child end
        end
    end
    return nil
end

task.spawn(function()
    local waited = 0
    while not net and waited < 15 do
        net = tryFindNet()
        if net then return end
        task.wait(0.5)
        waited = waited + 0.5
    end
    if not net then
        warn("[CZ SPEED BYPASS] Could not resolve a remote. GUI will still work.")
    end
end)

--==============================================================
-- LAGGER  (payload = {power, power, power, power} · coroutine per shot)
--==============================================================
local function fireLoop(token)
    while running and token == runToken do
        local v = power
        task.spawn(function()
            if not running or token ~= runToken then return end
            pcall(net.FireServer, net, {v, v, v, v})
        end)
        task.wait(FIRE_INTERVAL)
    end
end

local function startLagger()
    if running then return end

    local parsed = parsePower(powerBox.Text)
    if parsed then power = parsed end
    powerBox.Text = formatPower(power)
    refreshSlider()

    running = true
    runToken = runToken + 1
    local token = runToken
    setRunningVisuals(true)

    task.spawn(function()
        local waited = 0
        while running and token == runToken and not net and waited < NET_WAIT do
            task.wait(0.25)
            waited = waited + 0.25
        end
        if not running or token ~= runToken then return end
        if not net then
            warn("[CZ SPEED BYPASS] Remote never became available — stopping.")
            if stopLagger then stopLagger() end
            return
        end
        fireLoop(token)
    end)

    if queueSave then queueSave() end
end

stopLagger = function()
    if not running then return end
    running = false
    runToken = runToken + 1
    setRunningVisuals(false)
    if queueSave then queueSave() end
end

toggleLagger = function()
    if running then stopLagger() else startLagger() end
end

startBtn.Activated:Connect(toggleLagger)

local function beginCapture()
    if capturing then return end
    capturing = true
    refreshKeyBox()
end

local function endCapture()
    capturing = false
    refreshKeyBox()
end

local function overKeyBox(position)
    local a = keyBox.AbsolutePosition
    local s = keyBox.AbsoluteSize
    return position.X >= a.X and position.X <= a.X + s.X
        and position.Y >= a.Y and position.Y <= a.Y + s.Y
end

local function handleCapture(input)
    local kind = input.UserInputType
    if kind == Enum.UserInputType.MouseButton1 or kind == Enum.UserInputType.Touch then
        if not overKeyBox(input.Position) then endCapture() end
        return
    end
    if kind == Enum.UserInputType.Keyboard or string.find(kind.Name, "Gamepad") then
        local key = input.KeyCode
        if key == Enum.KeyCode.Unknown then return end
        if key == Enum.KeyCode.Escape or key == Enum.KeyCode.ButtonB then
            endCapture(); return
        end
        setBind("key", key.Name, friendlyKey(key.Name))
        endCapture()
        return
    end
    if kind == Enum.UserInputType.MouseButton2 then
        setBind("mouse", "MouseButton2", "RMB"); endCapture()
    elseif kind == Enum.UserInputType.MouseButton3 then
        setBind("mouse", "MouseButton3", "MMB"); endCapture()
    end
end

keyBox.Activated:Connect(function()
    if capturing then endCapture() else beginCapture() end
end)

local saveToken = 0

function saveConfig()
    if type(writefile) ~= "function" then return end
    local ok, data = pcall(function()
        return HttpService:JSONEncode({
            version = 16,
            name = TITLE_TEXT,
            power = power,
            powerText = powerBox.Text,
            keybind = { kind = bind.kind, code = bind.code, label = bind.label },
            bg = currentBg,
            walkSpeed = targetWalkSpeed,
            position = {
                x = math.floor(panel.Position.X.Offset),
                y = math.floor(panel.Position.Y.Offset),
            },
            expanded = expanded,
        })
    end)
    if ok and type(data) == "string" then pcall(writefile, CFG_FILE, data) end
end

function queueSave()
    if type(writefile) ~= "function" then return end
    saveToken = saveToken + 1
    local mine = saveToken
    task.delay(AUTOSAVE, function()
        if mine == saveToken then saveConfig() end
    end)
end

function loadConfig()
    if type(isfile) ~= "function" or type(readfile) ~= "function" then return false end
    local okFile, hasFile = pcall(isfile, CFG_FILE)
    if not hasFile and type(isfile) == "function" then
        -- migrate from legacy config if it exists
        local okLegacy, hasLegacy = pcall(isfile, LEGACY_CFG_FILE)
        if okLegacy and hasLegacy then
            pcall(function() writefile(CFG_FILE, readfile(LEGACY_CFG_FILE)) end)
            hasFile = true
        end
    end
    if not okFile or not hasFile then return false end

    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(CFG_FILE)) end)
    if not ok or type(data) ~= "table" then return false end

    local restored
    if type(data.powerText) == "string" then restored = parsePower(data.powerText) end
    if not restored and tonumber(data.power) then restored = parsePower(data.power) end
    if restored then power = restored end
    powerBox.Text = formatPower(power)
    refreshSlider()

    if type(data.keybind) == "table" then
        local kind  = data.keybind.kind
        local code  = data.keybind.code
        local label = data.keybind.label
        if kind == "none" then
            setBind("none", "None", "--", false)
        elseif kind == "mouse" and type(code) == "string" and Enum.UserInputType[code] then
            setBind("mouse", code, tostring(label or code), false)
        elseif type(code) == "string" and code ~= "" then
            local real = Enum.KeyCode[code]
            if real then setBind("key", code, tostring(label or friendlyKey(code)), false) end
        end
    end

    if data.bg then
        currentBg = tostring(data.bg)
        bgBox.Text = currentBg
        updateBg()
    end

    if tonumber(data.walkSpeed) then
        targetWalkSpeed = clamp(tonumber(data.walkSpeed), 1, 500)
        task.defer(applyWalkSpeed)
    end

    local pos = data.position
    if type(pos) == "table" and tonumber(pos.x) and tonumber(pos.y) then
        panel.Position = clampToScreen(tonumber(pos.x), tonumber(pos.y))
    end

    if data.expanded == true then setExpanded(true, true) end
    return true
end

function cacheIntroSong(option, allowDownload)
    if not option or not option.url or option.url == "" then return nil end
    if not (writefile and getcustomasset) then return nil end
    local fileName = option.file or ("Xlu0DuelsIntroSong_" .. tostring(option.name or "song") .. ".mp3")

    local function loadExisting()
        if introSongCache[fileName] then return introSongCache[fileName] end
        local hasFile = false
        pcall(function() hasFile = isfile and isfile(fileName) end)
        if hasFile then
            local ok = pcall(function() introSongCache[fileName] = getcustomasset(fileName) end)
            if ok and introSongCache[fileName] then return introSongCache[fileName] end
        end
        return nil
    end

    local cached = loadExisting()
    if cached then return cached end
    if allowDownload == false then return nil end
    if introSongDownloading[fileName] then
        local waitStart = tick()
        while introSongDownloading[fileName] and tick() - waitStart < 12 do task.wait(0.05) end
        cached = loadExisting()
        if cached then return cached end
    end

    introSongDownloading[fileName] = true
    local ok = pcall(function()
        local data = game:HttpGet(option.url)
        if data and #data > 0 then
            writefile(fileName, data)
            introSongCache[fileName] = getcustomasset(fileName)
        end
    end)
    introSongDownloading[fileName] = nil
    if ok and introSongCache[fileName] then return introSongCache[fileName] end
    return loadExisting()
end

    end)
end

    end)

        IntroTitleSweep.Name = "IntroTitleSweep"
        IntroTitleSweep.TextColor3 = PURPLE
        IntroTitleSweep.TextTransparency = 0
        IntroTitleSweep.TextStrokeTransparency = 1
        IntroTitleSweep.ZIndex = IntroTitle.ZIndex + 1
        IntroTitleSweep.Parent = AdaptIntro
        _G.CzSpeedBypassIntroTitleSweep = IntroTitleSweep

        local IntroSweepGradient = Instance.new("UIGradient")
        IntroSweepGradient.Rotation = 12
        IntroSweepGradient.Offset = Vector2.new(-1.25, 0)
        IntroSweepGradient.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0.00, 1),
            NumberSequenceKeypoint.new(0.42, 1),
            NumberSequenceKeypoint.new(0.50, 0.08),
            NumberSequenceKeypoint.new(0.58, 1),
            NumberSequenceKeypoint.new(1.00, 1),
        })
        IntroSweepGradient.Parent = IntroTitleSweep

        task.spawn(function()
            while IntroTitleSweep and IntroTitleSweep.Parent do
                IntroSweepGradient.Offset = Vector2.new(-1.25, 0)
                local sweep = TweenService:Create(
                    IntroSweepGradient,
                    TweenInfo.new(1.2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut),
                    { Offset = Vector2.new(1.25, 0) }
                )
                sweep:Play()
                sweep.Completed:Wait()
                task.wait(0.8)
            end
        end)

        new("TextLabel", {
            Name = "TapAnywhereShadow", ZIndex = 1002,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.new(0.5, 3, 0.42, 61),
            Size = UDim2.new(0, 300, 0, 38), BackgroundTransparency = 1,
            Text = "TAP ANYWHERE TO SKIP",
            TextColor3 = Color3.fromRGB(0, 0, 0), TextTransparency = 0.18,
            TextSize = 18, Font = Enum.Font.GothamBlack, Parent = AdaptIntro,
        })

        local TapAnywhereLabel = new("TextLabel", {
            Name = "TapAnywhere", ZIndex = 1003,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.new(0.5, 0, 0.42, 58),
            Size = UDim2.new(0, 300, 0, 38), BackgroundTransparency = 1,
            Text = "TAP ANYWHERE TO SKIP",
            TextColor3 = WHITE, TextStrokeColor3 = Color3.fromRGB(0, 0, 0),
            TextStrokeTransparency = 0.45, TextSize = 18,
            Font = Enum.Font.GothamBlack, Parent = AdaptIntro,
        })

        new("TextLabel", {
            Name = "DiscordInviteShadow", ZIndex = 1002,
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.new(0.5, 3, 0.42, 85),
            Size = UDim2.new(0, 250, 0, 34), BackgroundTransparency = 1,
            Text = "discord.gg/93czvm9Vrt",
            TextColor3 = Color3.fromRGB(0, 0, 0), TextTransparency = 0.2,
            TextSize = 16, Font = Enum.Font.GothamBlack, Parent = AdaptIntro,

            Text = "discord.gg/93czvm9Vrt",

            TextStrokeTransparency = 0.5, TextSize = 16,
            Font = Enum.Font.GothamBlack, Parent = AdaptIntro,
        })

        local introTextScaleValue = Xlu0AutoMobile and 0.82 or 1.00
        for _, obj in ipairs({ IntroTitle, IntroTitleSweep, TapAnywhereLabel, DiscordInviteLabel }) do
            local sc = Instance.new("UIScale")
            sc.Name = "Xlu0AutoDeviceScale"
            sc.Scale = introTextScaleValue
            sc.Parent = obj
        end
        for _, shadowName in ipairs({ "TapAnywhereShadow", "DiscordInviteShadow" }) do
            local sh = AdaptIntro:FindFirstChild(shadowName)
            if sh then
                local sc = Instance.new("UIScale")
                sc.Name = "Xlu0AutoDeviceScale"
                sc.Scale = introTextScaleValue
                sc.Parent = sh
            end
        end

            Name = "TapCatcher", ZIndex = 1004,
            Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1,
            Text = "", AutoButtonColor = false, Parent = AdaptIntro,
        })

        local introSkipped = false
        local TapAnywhere = AdaptIntro:FindFirstChild("TapAnywhere")

        local function skipIntro()
            if introSkipped then return end
            introSkipped = true
            stopIntroPlayback()
            _G.CzSpeedBypassIntroTitleSweep = nil

            tw(AdaptIntro, 0.4, { BackgroundTransparency = 1 })
            for _, child in ipairs(AdaptIntro:GetChildren()) do
                if child:IsA("ImageLabel") then
                    tw(child, 0.3, { ImageTransparency = 1 })
                elseif child:IsA("TextLabel") then
                    tw(child, 0.3, { TextTransparency = 1 })
                elseif child:IsA("Frame") then
                    tw(child, 0.3, { BackgroundTransparency = 1 })
                    for _, sub in ipairs(child:GetDescendants()) do
                        if sub:IsA("ImageLabel") then tw(sub, 0.3, { ImageTransparency = 1 }) end
                        if sub:IsA("TextLabel") then tw(sub, 0.3, { TextTransparency = 1 }) end
                    end
                end
            end

            task.delay(0.45, function()
                if revealPanel then revealPanel() end
                if IntroGui then IntroGui:Destroy() end
            end)
        end

            task.wait(0.6)
            if introSkipped then return end

            TojiCutoutStage.Visible = true
            local tojiScale = TojiCutoutStage:FindFirstChild("UIScale")
            if tojiScale then
                tojiScale.Scale = 0.78
                tw(tojiScale, 0.6, { Scale = 1 }, Enum.EasingStyle.Back)
            end

            local pieceOffsets = {
                {-42,-26,-10}, {38,-34,9}, {-34,18,-7}, {30,28,8},
                {-48,6,-11},   {46,10,10}, {-25,38,-6}, {28,42,7},
                {-18,-44,-5},  {20,-38,5},
            }

            for i = 1, 10 do
                if introSkipped then return end
                local data = pieceOffsets[i]
                local piece = TojiCutoutStage:FindFirstChild("TojiPiece" .. i)
                if piece then
                    piece.Position = UDim2.new(0.5, data[1], 0.5, data[2])
                    piece.Rotation = data[3]
                    piece.ImageTransparency = 1
                    tw(piece, 0.42, {
                        Position = UDim2.new(0.5, 0, 0.5, 0),
                        Rotation = 0, ImageTransparency = 0,
                    }, Enum.EasingStyle.Back)
                end
                for _, child in ipairs(TojiCutoutStage:GetChildren()) do
                    if child.Name == "TojiSeamFill" .. i then
                        child.Position = UDim2.new(0.5, data[1], 0.5, data[2])
                        child.Rotation = data[3]
                        child.ImageTransparency = 1
                        tw(child, 0.42, {
                            Position = UDim2.new(0.5, 0, 0.5, 0),
                            Rotation = 0, ImageTransparency = 0,
                        }, Enum.EasingStyle.Back)
                    end
                end
                task.wait(0.065)
            end

            tw(tojiScale, 0.16, { Scale = 1.06 }, Enum.EasingStyle.Quad)
            task.wait(0.16)
            tw(tojiScale, 0.24, { Scale = 1 }, Enum.EasingStyle.Back)

            task.spawn(function()
                while not introSkipped and TojiCutoutStage.Parent do
                    tw(TojiCutoutStage, 1.15, {
                        Rotation = 0.8, Position = UDim2.new(0.5, 0, 0.5, -5),
                    }, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
                    task.wait(1.15)
                    if introSkipped or not TojiCutoutStage.Parent then return end
                    tw(TojiCutoutStage, 1.15, {
                        Rotation = -0.8, Position = UDim2.new(0.5, 0, 0.5, 5),
                    }, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
                    task.wait(1.15)
                end
            end)

            task.wait(0.3)
            if introSkipped then return end

            local shine = TojiCutoutStage:FindFirstChild("TojiSwordShine")
            if shine then
                tw(shine, 0.4, { ImageTransparency = 0.2 })
                task.wait(0.3)
                if not introSkipped then tw(shine, 0.3, { ImageTransparency = 0.7 }) end
            end

            task.wait(0.5)
            if introSkipped then return end

            task.spawn(function()
                while not introSkipped do
                    tw(TapAnywhere, 0.8, { TextTransparency = 0.5 })
                    task.wait(0.8)
                    if introSkipped then return end
                    tw(TapAnywhere, 0.8, { TextTransparency = 0 })
                    task.wait(0.8)
                end
            end)

            task.wait(INTRO_HOLD)
            if not introSkipped then skipIntro() end
        end)
    end)
end
