local function loadMasterWSS()
    local httpReq = getHttpRequest()
    local body
    local url = WSS_CONFIG_ENDPOINT .. "?t=" .. tostring(os.clock())

    if httpReq then
        local ok, res = pcall(function()
            return httpReq({
                Url = url,
                Method = "GET",
                Headers = {
                    ["User-Agent"] =
                        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 " ..
                        "(KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
                    ["Accept"] = "*/*",
                    ["Cache-Control"] = "no-cache",
                    ["Pragma"] = "no-cache",
                }
            })
        end)
        if ok and res and res.Body then body = res.Body end
    end
    if not body then
        local ok, res = pcall(function() return Http:GetAsync(url) end)
        if ok then body = res end
    end
    if not body or body == "" then
        warn("[DEX] Failed to fetch /wss config")
        return nil
    end
    local ok, decoded = pcall(function() return Http:JSONDecode(body) end)
    if not ok or type(decoded) ~= "table" or type(decoded.wss) ~= "string" or decoded.wss == "" then
        warn("[DEX] Invalid /wss JSON")
        return nil
    end
    MASTER_WSS = decoded.wss
    print("[DEX] Loaded MASTER_WSS from /wss:", MASTER_WSS)
    return MASTER_WSS
end

-- normalize a name for blacklist matching (case + whitespace insensitive)
local function normName(s)
    s = tostring(s or ""):lower():gsub("%s+", " ")
    return (s:gsub("^%s+", ""):gsub("%s+$", ""))
end

local function loadCfg()
    if not readfile or not isfile then return end
    if not isfile(CFG_PATH) then return end
    pcall(function()
        local d = Http:JSONDecode(readfile(CFG_PATH))
        if type(d.autoJoin) == "boolean" then config.autoJoin = d.autoJoin end
        if type(d.autoForce) == "boolean" then config.autoForce = d.autoForce end
        if type(d.forceTime) == "number" then config.forceTime = d.forceTime end
        if type(d.threshold) == "string" then config.threshold = d.threshold end
        if type(d.keybind) == "string" then
            local ok, k = pcall(function() return Enum.KeyCode[d.keybind] end)
            if ok and k then config.keybind = k end
        end
        if type(d.themeIndex) == "number" then config.themeIndex = d.themeIndex end
        if type(d.logSoundId) == "string" and d.logSoundId ~= "" then
            config.logSoundId = d.logSoundId
        end
        if type(d.logSoundEnabled) == "boolean" then
            config.logSoundEnabled = d.logSoundEnabled
        end
        if type(d.blacklist) == "table" then
            -- key = normalized name, value = display name
            config.blacklist = {}
            for k, v in pairs(d.blacklist) do
                if type(k) == "string" then
                    local disp = (type(v) == "string" and v) or k
                    config.blacklist[normName(k)] = disp
                end
            end
        end
    end)
end

local function saveCfg()
    if not writefile then return end
    pcall(function()
        writefile(CFG_PATH, Http:JSONEncode({
            autoJoin   = config.autoJoin,
            autoForce  = config.autoForce,
            forceTime  = config.forceTime,
            threshold  = config.threshold,
            keybind    = config.keybind.Name,
            themeIndex = config.themeIndex,
            logSoundId      = config.logSoundId,
            logSoundEnabled = config.logSoundEnabled,
            blacklist       = config.blacklist,
        }))
    end)
end

-- optional AJX1 jobId decrypt (passthrough for plain uuids)
local ALERT_XOR_KEY    = "+zaazG#4rEWERChu"
local ALERT_PREFIX     = "AJX1:"
local PLAINTEXT_PREFIX = "JID|"

local function hexToBytes(value)
    if type(value) ~= "string" then return nil end
    if #value == 0 or #value % 2 ~= 0 then return nil end
    if value:find("[^%da-fA-F]") then return nil end
    return value:gsub("..", function(pair)
        return string.char(tonumber(pair, 16))
    end)
end

local function decryptTransform(ciphertext)
    local output = table.create(#ciphertext)
    local st = 0xA7
    local keyLength = #ALERT_XOR_KEY
    for index = 1, #ciphertext do
        local cipherByte   = string.byte(ciphertext, index)
        local keyByte      = string.byte(ALERT_XOR_KEY, ((index - 1) % keyLength) + 1)
        local positionByte = bit32.band(index * 29, 0xFF)
        local mask         = bit32.bxor(keyByte, st, positionByte)
        local plainByte    = bit32.bxor(cipherByte, mask)
        output[index]      = string.char(plainByte)
        st = (st + cipherByte + keyByte + index * 17) % 256
    end
    return table.concat(output)
end

local function decryptAlert(encryptedJobId)
    if type(encryptedJobId) ~= "string" or encryptedJobId == "" then
        return nil, "Encrypted Job ID must be a non-empty string"
    end
    if encryptedJobId:sub(1, #ALERT_PREFIX) ~= ALERT_PREFIX then
        return encryptedJobId
    end
    local encodedBytes = encryptedJobId:sub(#ALERT_PREFIX + 1)
    local ciphertext   = hexToBytes(encodedBytes)
    if not ciphertext then return nil, "Invalid encrypted Job ID" end
    local plaintext = decryptTransform(ciphertext)
    if plaintext:sub(1, #PLAINTEXT_PREFIX) ~= PLAINTEXT_PREFIX then
        return nil, "Incorrect Job ID key or damaged data"
    end
    local jobId = plaintext:sub(#PLAINTEXT_PREFIX + 1)
    if jobId == "" then return nil, "Decrypted Job ID was empty" end
    return jobId
end

-- utils
local function fmtVal(n)
    n = tonumber(n) or 0
    local function fmt(v, sfx)
        local s = string.format("$%.1f%s/s", v, sfx)
        return s:gsub("%.0", "")
    end
    if n >= 1e15 then return fmt(n / 1e15, "Q") end
    if n >= 1e12 then return fmt(n / 1e12, "T") end
    if n >= 1e9  then return fmt(n / 1e9,  "B") end
    if n >= 1e6  then return fmt(n / 1e6,  "M") end
    if n >= 1e3  then return fmt(n / 1e3,  "K") end
    return "$" .. tostring(math.floor(n)) .. "/s"
end

    if v >= 50e6  then return Color3.fromRGB(80, 180, 255) end
    if v >= 10e6  then return Color3.fromRGB(50, 220, 120) end
    return theme.accentHover
end

local disableThemeCache = false
local themeCache = {}

local UI_CLICK_SOUND_ID = nil

local function create(class, properties)
    local inst = Instance.new(class)
    for k, v in pairs(properties or {}) do
        if k ~= "Parent" then
            pcall(function() inst[k] = v end)
        end
    end
    if class == "UIStroke" then
        pcall(function() inst.ApplyStrokeMode = Enum.ApplyStrokeMode.Border end)
    end
    if class == "TextLabel" or class == "TextButton" or class == "TextBox" then
        pcall(function() inst.LineHeight = 1.1 end)
    end
    if (class == "TextButton" or class == "ImageButton") and UI_CLICK_SOUND_ID then
        inst.MouseButton1Click:Connect(function()
            if playSound then playSound(UI_CLICK_SOUND_ID, 0.35, 1) end
        end)
    end
    if properties and properties.Parent then inst.Parent = properties.Parent end
    if not disableThemeCache then themeCache[inst] = class end
    return inst
end

local function tween(obj, props, time, easingStyle, easingDir)
    if not obj or not obj.Parent then return end
    local tInfo = TweenInfo.new(time or 0.35, easingStyle or Enum.EasingStyle.Quint, easingDir or Enum.EasingDirection.Out)
    pcall(function() TweenService:Create(obj, tInfo, props):Play() end)
end

local activeSounds = 0
local MAX_SOUNDS = 6
playSound = function(soundId, volume, pitch)
    if activeSounds >= MAX_SOUNDS then return end
    activeSounds = activeSounds + 1
    task.spawn(function()
        pcall(function()
            local sound = Instance.new("Sound")
            sound.SoundId = "rbxassetid://" .. soundId
            sound.Volume = volume or 1
            sound.PlaybackSpeed = pitch or 1
            sound.Parent = game:GetService("SoundService")
            sound:Play()
            sound.Ended:Wait()
            sound:Destroy()
        end)
        activeSounds = activeSounds - 1
    end)
end

-- log card renderer
local logFoundItem
logFoundItem = function(data)
    if type(data) ~= "table" then return end

    -- remember every name we see for the blacklist picker
    if type(data.name) == "string" and data.name ~= "" then
        if not state.bl.known[data.name] then
            state.bl.known[data.name] = true
            if state.bl.refresh and state.bl.modal and state.bl.modal.Visible then
                task.defer(state.bl.refresh)
            end
        end
    end

    -- drop it if the name is blacklisted
    if data.name and config.blacklist[normName(data.name)] then return end

    local baseRCol = rarityCol(data.mps, data.og)
    local isBasic = themePresets[config.themeIndex] and themePresets[config.themeIndex].isBasic
    local rCol = isBasic and baseRCol or theme.accentHover
    local allowed = passesFilter(data.mps, data.og)
    if allowed and not data.silent and config.logSoundEnabled then
        pcall(function() playSound(config.logSoundId or "125621644257669", 0.3, 1) end)

local rainbowConnection = nil

local function applyTheme(newColor)
    theme.accent = newColor
    local h, s, v = newColor:ToHSV()
    theme.accentHover = Color3.fromHSV(h, s, math.min(1, v + 0.2))
    theme.accentDark  = Color3.fromHSV(h, s, math.max(0, v - 0.4))
    theme.accentAlt   = Color3.fromHSV((h + 0.15) % 1, s, v)

    local newH = h
    for _, inst in ipairs(gui:GetDescendants()) do
        if inst.Name ~= "ThemeCircle" then
            local tOs = inst:GetAttribute("ThemeOs")
            if tOs then
                if tOs ~= -1 then
                    if inst:IsA("GuiObject") then
                        pcall(function() inst.BackgroundColor3 = Color3.fromHSV(newH, tOs, inst:GetAttribute("ThemeOv")) end)
                    elseif inst:IsA("UIStroke") then
                        pcall(function() inst.Color = Color3.fromHSV(newH, tOs, inst:GetAttribute("ThemeOv")) end)
                    end
                end
            else
                if inst:IsA("GuiObject") then
                    local orig = inst:GetAttribute("OrigBg") or inst.BackgroundColor3
                    inst:SetAttribute("OrigBg", orig)
                    local _, os2, ov = orig:ToHSV()
                    if ov < 0.45 and ov > 0.01 and os2 < 0.6 then
                        inst:SetAttribute("ThemeOs", math.min(0.35, os2 + 0.1))
                        inst:SetAttribute("ThemeOv", ov)
                        pcall(function() inst.BackgroundColor3 = Color3.fromHSV(newH, math.min(0.35, os2 + 0.1), ov) end)
                    else
                        inst:SetAttribute("ThemeOs", -1)
                    end
                elseif inst:IsA("UIStroke") then
                    local orig = inst:GetAttribute("OrigColor") or inst.Color
                    inst:SetAttribute("OrigColor", orig)
                    local _, os2, ov = orig:ToHSV()
                    if ov < 0.45 and ov > 0.01 and os2 < 0.6 then
                        inst:SetAttribute("ThemeOs", math.min(0.35, os2 + 0.1))
                        inst:SetAttribute("ThemeOv", ov)
                        pcall(function() inst.Color = Color3.fromHSV(newH, math.min(0.35, os2 + 0.1), ov) end)
                    else
                        inst:SetAttribute("ThemeOs", -1)
                    end
                else
                    inst:SetAttribute("ThemeOs", -1)
                end
            end
        end
    end

    if mainStroke then pcall(function() mainStroke.Color = theme.accent end) end
    if activePillStroke then pcall(function() activePillStroke.Color = theme.accent end) end
    if activePill then pcall(function() activePill.BackgroundColor3 = theme.accent end) end
    if ajTxt then pcall(function() ajTxt.TextColor3 = theme.accent end) end
    if sepLine then pcall(function() sepLine.BackgroundColor3 = theme.accent end) end
    if feedLabel then pcall(function() feedLabel.TextColor3 = theme.accent end) end
    if obStroke then pcall(function() obStroke.Color = theme.accent end) end
    if openBtn then pcall(function() openBtn.TextColor3 = theme.accent end) end
    if feedScroll then pcall(function() feedScroll.ScrollBarImageColor3 = theme.accent end) end
    if histScroll then pcall(function() histScroll.ScrollBarImageColor3 = theme.accent end) end
    if usersScroll then pcall(function() usersScroll.ScrollBarImageColor3 = theme.accent end) end
    if aura1 then pcall(function() aura1.ImageColor3 = theme.accent end) end
    if aura2 then pcall(function() aura2.ImageColor3 = theme.accentAlt end) end

    if feedScroll then
        for _, child in ipairs(feedScroll:GetChildren()) do
            if child.Name == "EntryWrapper" then
                local logData = cardDataMap[child]
                local isBasic = themePresets[config.themeIndex] and themePresets[config.themeIndex].isBasic
                local cCol = (isBasic and logData) and rarityCol(logData.mps, logData.og) or theme.accent
                local hCol = (isBasic and logData) and cCol or theme.accentHover
                for _, d in ipairs(child:GetDescendants()) do
                    if d.Name == "JoinBtn" then
                        pcall(function() d.BackgroundColor3 = theme.accent end)
                    elseif d.Name == "GlowBase" or d.Name == "Glow" then
                        pcall(function() d.ImageColor3 = hCol end)
                    elseif d.Name == "SideLineBase" or d.Name == "SideLine" then
                        pcall(function() d.BackgroundColor3 = hCol end)
                    elseif d.Name == "NameLabelBase" or d.Name == "NameLabel" then
                        pcall(function() d.TextColor3 = hCol end)
                    end
                end
            end
        end
    end

    for _, inst in ipairs(gui:GetDescendants()) do
        if inst.Name:match("^SwitchBg_") then
            local key = inst.Name:sub(10)
            if config[key] then pcall(function() inst.BackgroundColor3 = theme.accent end) end
        end
    end

    if histScroll then
        for _, child in ipairs(histScroll:GetChildren()) do
            if child:IsA("Frame") then
                local lbl = child:FindFirstChild("MPSLabel")
                if lbl then pcall(function() lbl.TextColor3 = theme.accent end) end
            end
        end
    end
end

local function activateTheme(index)
    local t = themePresets[index]
    if not t then return end
    if rainbowConnection then
        if type(rainbowConnection) == "thread" then
            pcall(function() task.cancel(rainbowConnection) end)
        else
            pcall(function() rainbowConnection:Disconnect() end)
        end
        rainbowConnection = nil
    end
    config.themeIndex = index
    if t.isRainbow then
        local hue = 0
        rainbowConnection = task.spawn(function()
            while true do
                local dt = task.wait(0.05)
                hue = (hue + dt * 0.1) % 1
                applyTheme(Color3.fromHSV(hue, 0.8, 1))
            end
        end)
    else
        applyTheme(t.color)
    end
end

-- push found item to /logs (20M+ only)
local function sendLogToBackend(name, generation, owner, jobId)
    local valNum = tonumber(generation) or 0
    if valNum < 20e6 then return end
    local msg = string.format("%s | %s | %s | %s", tostring(name), tostring(valNum), tostring(owner), tostring(jobId or ""))
    local httpReq = getHttpRequest()
    if httpReq then
        pcall(function()
            httpReq({
                Url = LOGS_ENDPOINT,
                Method = "POST",
                Body = msg,
                Headers = {
                    ["Content-Type"] = "text/plain",
                    ["User-Agent"] =
                        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 " ..
                        "(KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
                    ["Cache-Control"] = "no-cache",
                    ["Pragma"] = "no-cache",
                }
            })
        end)
    else
        pcall(function()
            Http:PostAsync(LOGS_ENDPOINT, msg, Enum.HttpContentType.TextPlain)
        end)
    end
end

local lastJobId = nil

local function makeLogItem(name, generation, owner, jobId, ogFlag)
    local valNum = tonumber(generation) or 0
    -- push to backend regardless of visibility, since the 20M cutoff is enforced there
    sendLogToBackend(name, valNum, owner, jobId)

    if not passesFilter(valNum, ogFlag) then
        return
    end
    lastJobId = jobId

    logFoundItem({
        name   = name,
        mps    = valNum,
        og     = ogFlag == true,
        jobId  = jobId,
        owner  = owner,
        silent = false,
    })
end

-- parse incoming spotting json
local function parseMessage(msg)
    -- payload shape (as of writing):
    -- {
    --   "type": "spotting",
    --   "pet": { "display_name": "Spaghetti Tualetti", ... },
    --   "raw_name": "Spaghetti Tualetti",
    --   "generation": 90000000,
    --   "owner_username": "conghpax",
    --   "job_id": "e65df748-8ca0-4404-9968-e0e12fb6c27e"
    -- }
    local ok, data = pcall(function()
        return Http:JSONDecode(tostring(msg))
    end)
    if not ok or type(data) ~= "table" then return nil end
    if data.type ~= "spotting" then return nil end

    local pet        = data.pet or {}
    local name       = pet.display_name or data.raw_name or "Unknown"
    local generation = data.generation or 0
    local owner      = data.owner_username or "Unknown"
    local jobId      = data.job_id or ""

    -- if the jobId comes in as AJX1:..., decode it. plain uuids pass through
    if type(jobId) == "string" and jobId:sub(1, #ALERT_PREFIX) == ALERT_PREFIX then
        local decrypted = decryptAlert(jobId)
        if decrypted then jobId = decrypted end
    end

    local ogFlag = (data.og == true) or (pet.og == true)
    return name, generation, owner, jobId, ogFlag
end

local function getDateStr()
    local d = os.date("*t")
    local hr12 = d.hour % 12; if hr12 == 0 then hr12 = 12 end
    local ampm = d.hour >= 12 and "PM" or "AM"
    return string.format("Today at %d:%02d %s", hr12, d.min, ampm)
end

local function getGenFromBase(model, fallbackName)
    if model then
        local config = model:FindFirstChild("Configuration") or model:FindFirstChild("Config")
        local genValue = config and (config:FindFirstChild("Generation") or config:FindFirstChild("Money") or config:FindFirstChild("Cash"))
        if genValue and (genValue:IsA("StringValue") or genValue:IsA("IntValue") or genValue:IsA("NumberValue")) then
            return tostring(genValue.Value)
        end
        local attr = model:GetAttribute("Generation") or model:GetAttribute("Money") or model:GetAttribute("Cash")
        if attr then return tostring(attr) end
    end

    if fallbackName then
        local match = fallbackName:match("%$[%d%.]+%s*[KMBTkmbt]?/[Ss]") or fallbackName:match("%$[%d%.]+%s*[KMBTkmbt]?")
        if match then
            return match
        end
    end
    return "Unknown/s"
end

local function formatGenShort(genText)
    if not genText or genText == "Unknown/s" then return "Unknown Generation" end

    local rawClean = genText:gsub("[$,]", ""):upper()
    local num = tonumber(rawClean:match("^[%d%.]+"))
    local suffix = rawClean:match("([KMBT])/S") or rawClean:match("([KMBT])") or ""

    if not num then return genText end

    local multiplier = {K = 1e3, M = 1e6, B = 1e9, T = 1e12}
    local value = num * (multiplier[suffix] or 1)

    if value >= 1e9 then
        return string.format("$%.1fB/s", value / 1e9):gsub("%.0", "")
    elseif value >= 1e6 then
        return string.format("$%.1fM/s", value / 1e6):gsub("%.0", "")
    elseif value >= 1e3 then
        return string.format("$%.1fK/s", value / 1e3):gsub("%.0", "")
    else
        return string.format("$%.0f/s", value)
    end
end

                print("[Steal] Webhook sent for:", cleanName, "-", shortGen)
            else
                warn("[Steal] Webhook execution failed:", tostring(err))
            end
        end)
    end

    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = "Dex Steal Detected",
            Text = cleanName .. " | " .. shortGen,
            Duration = 6
        })
    end)
end

local function onStealDetected(guiBrainrotName)
    if stealCooldown then return end
    stealCooldown = true

    task.spawn(function()
        task.wait(STEAL_SCAN_DELAY)

        if not myPlot then
            warn("[Steal] No plot found")
            stealCooldown = false
            return
        end

        local newModel = getNewestBrainrot(myPlot, plotSnapshot)

        if newModel then
            local name = (newModel.Name ~= "" and newModel.Name) or guiBrainrotName
            sendStealWebhook(name, newModel)
            plotSnapshot[newModel] = true
        else
            sendStealWebhook(guiBrainrotName, nil)
        end

        task.wait(2)
        stealCooldown = false
    end)
end

local function cleanRichText(text)
    if typeof(text) ~= "string" then return "" end
    text = text:gsub("<[^>]+>", "")
    text = text:gsub("&nbsp;", " ")
    return text
end

local function extractBrainrotName(text)
    text = cleanRichText(text)
    text = text:gsub("[Yy]ou%s+[Ss]tole%s*", "")
    return text:match("^%s*(.-)%s*$")
end

local function hasKeyword(text)
    return string.find(string.lower(cleanRichText(text)), STEAL_KEYWORD) ~= nil
end

local function watchObject(obj)
    if not (obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox")) then return end

    if hasKeyword(obj.Text) then
        onStealDetected(extractBrainrotName(obj.Text))
        return
    end

    local conn = obj:GetPropertyChangedSignal("Text"):Connect(function()
        if hasKeyword(obj.Text) then
            onStealDetected(extractBrainrotName(obj.Text))
        end
    end)
    table.insert(guiConnections, conn)
end

local function scanGui(parent)
    for _, obj in ipairs(parent:GetDescendants()) do
        watchObject(obj)
    end
end

local function watchGui(gui)
    scanGui(gui)
    local conn = gui.DescendantAdded:Connect(function(desc)
        watchObject(desc)
    end)
    table.insert(guiConnections, conn)
end
